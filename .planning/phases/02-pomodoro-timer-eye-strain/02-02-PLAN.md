Add four new async methods to NotificationService, following the `sendTestNotification()` pattern:

**1. Work complete notification** (POMO-05):

```swift
func sendWorkCompleteNotification(includeEyeStrain: Bool) async {
    let content = UNMutableNotificationContent()
    content.title = "Time for a Break!"
    if includeEyeStrain {
        content.body = "Great focus! Look at something 20 feet away and stretch."
    } else {
        content.body = "Great focus! Rest your eyes and stretch."
    }
    content.sound = .default

    let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 1, repeats: false)
    let request = UNNotificationRequest(
        identifier: "work-complete",
        content: content,
        trigger: trigger
    )
    try? await center.add(request)
}
```

The `includeEyeStrain` parameter satisfies EYE-02: when the eye-strain reminder was suppressed because a break was within 3 minutes (per D-04), the "look away" copy is folded into the break notification.

```swift
func sendEyeStrainNotification() async {

    try? await center.add(request)
}
```

Use unique identifier per eye-strain notification (UUID suffix) so multiple can queue. EYE-03 (dismiss) is handled automatically by macOS notification center — all notifications are dismissible by default.

**3. Eye-strain timer management:**

Start eye-strain timer when work begins (in `start()` method):

```swift
private func startEyeStrainTimer() {
    eyeStrainTimer?.invalidate()
    eyeStrainStartDate = Date()
    eyeStrainSuppressed = false
    // Fire every 20 minutes
    eyeStrainTimer = Timer.scheduledTimer(withTimeInterval: 20 * 60, repeats: true) { [weak self] _ in
        Task { @MainActor in
            await self?.handleEyeStrainTick()
        }
    }
}
```

Stop eye-strain timer when work ends, pause, or stop:

```swift
private func stopEyeStrainTimer() {
    eyeStrainTimer?.invalidate()
    eyeStrainTimer = nil
    eyeStrainStartDate = nil
}
```

**4. Eye-strain tick handler** (D-04: suppress within 3 minutes of break):

```swift
private func handleEyeStrainTick() async {
    // D-04: If break is starting within 3 minutes, suppress and fold into break notification
    guard state == .working, let endDate else { return }
    let timeUntilBreak = endDate.timeIntervalSinceNow
    if timeUntilBreak <= 180 {  // 3 minutes = 180 seconds
        eyeStrainSuppressed = true
        return  // Notification will be folded into break notification
    }
    // Fire standalone eye-strain notification (EYE-01)
    await notificationService?.sendEyeStrainNotification()
}
```
