Create `Services/NotificationService.swift` — an `@Observable` class that wraps `UNUserNotificationCenter`:

```swift
import Foundation
import UserNotifications
import Observation

@Observable
@MainActor
final class NotificationService {
    var authorizationStatus: UNAuthorizationStatus = .notDetermined

    private let center = UNUserNotificationCenter.current()

    init() {
        // Check status on creation — non-blocking
        Task {
            await checkStatus()
        }
    }

    func checkStatus() async {
        let settings = await center.notificationSettings()
        authorizationStatus = settings.authorizationStatus
    }

    func requestPermission() async {
        do {
            let granted = try await center.requestAuthorization(options: [.alert, .sound])
            await checkStatus()
        } catch {
            await checkStatus()
        }
    }

    func sendTestNotification() async {
        let content = UNMutableNotificationContent()
        content.title = "PomoHydro"
        content.body = "Notifications are working! You'll receive reminders here."
        content.sound = .default

        let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 1, repeats: false)
        let request = UNNotificationRequest(
            identifier: UUID().uuidString,
            content: content,
            trigger: trigger
        )

        do {
            try await center.add(request)
        } catch {
            // Notification scheduling failed — permission may have been revoked
            await checkStatus()
        }
    }
}
```

Key design decisions:

- `@MainActor` isolation: All properties and methods run on main actor since views observe them. Prevents Swift 6 concurrency warnings.
- `authorizationStatus` is `@Observable` — SwiftUI views re-render when it changes.
- `requestPermission()` calls `checkStatus()` after to update the observed property regardless of grant/deny.
- `sendTestNotification()` schedules a 1-second delayed notification to verify infrastructure works.
- Uses `.alert` and `.sound` options (per D-03, no badge needed for menu bar app).
